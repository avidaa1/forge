const REDIRECTS=[
  "Not sure I've got that one — try asking about a product, payment, or how Forge routes orders.",
  "That's outside what's on this manifest. Ask about the catalog, or how checkout routes.",
  "Didn't land. Try 'what's for sale' or 'how do I pay'."
];
function scoreKB(msg){
  const m=msg.toLowerCase();let best=null,bestScore=0;
  KB.forEach(row=>{
    row.k.forEach(kw=>{
      if(m.includes(kw)){const s=kw.length;if(s>bestScore){bestScore=s;best=row;}}
    });
  });
  return {best,bestScore};
}
function chatMatch(msg){
  const {best,bestScore}=scoreKB(msg);
  if(best&&bestScore>=3)return {text:best.a};
  return {text:REDIRECTS[Math.floor(Math.random()*REDIRECTS.length)]};
}
function addMsg(text,cls){
  const div=document.createElement('div');div.className='cmsg '+cls;div.textContent=text;
  const m=document.getElementById('chat-msgs');m.appendChild(div);m.scrollTop=9999;
}
function showTyping(){
  const t=document.createElement('div');t.className='typing-indicator';t.id='typing';
  t.innerHTML='<span></span><span></span><span></span>';
  const m=document.getElementById('chat-msgs');m.appendChild(t);m.scrollTop=9999;return t;
}
let chatOpen=false;
function toggleChat(){
  chatOpen=!chatOpen;
  document.getElementById('chat-box').classList.toggle('open',chatOpen);
  if(chatOpen&&document.getElementById('chat-msgs').children.length===0){
    setTimeout(()=>addMsg('Forge Desk online. Ask about a product, payment routing, or the catalog.','ai'),300);
  }
  if(chatOpen)document.getElementById('chat-input').focus();
}
function chatSend(){
  const inp=document.getElementById('chat-input');const msg=inp.value.trim();if(!msg)return;
  addMsg(msg,'user');inp.value='';
  const typing=showTyping();
  setTimeout(()=>{
    typing.remove();
    const res=chatMatch(msg);
    addMsg(res.text,'ai');
  },350+Math.random()*300);
}
function quickAsk(q){document.getElementById('chat-input').value=q;chatSend();}
document.getElementById('chat-input').addEventListener('keydown',e=>{if(e.key==='Enter')chatSend();});
